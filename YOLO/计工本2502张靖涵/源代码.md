results = model.train(

# 3.图片推理xunlian.py程序代码:

```
import cv2
# 引入YOLO模型
from ultralytics import YOLO
# 打开图像
img_path = "d:/xunlian/img.jpg"  # 图像保存路径
# 打开图像
img = cv2.imread(filename=img_path)
# 加载模型
model = YOLO(model="yolo11n.pt")  # 图像保存路径
# 正向推理
res = model(img)
# 绘制推理结果
annotated_img = res[0].plot()
# 显示图像
cv2.imshow(winname="YOLO11", mat=annotated_img)
# 等待时间
cv2.waitKey(delay=10000)
# 绘制推理结果
cv2.imwrite(filename="jieguo.jpeg", img=annotated_img)
```

# 4.视频推理shipinxunlian.py程序代码：

```
import cv2
from ultralytics import YOLO
# 加载模型
model = YOLO(model="yolo11x.pt")
# 视频文件
video_path = "d:/xunlian/chepai.mp4"
# 打开视频
cap = cv2.VideoCapture(video_path)
while cap.isOpened():
    # 获取图像
    res, frame = cap.read()
    # 如果读取成功
    if res:
        # 正向推理
        results = model(frame)
        # 绘制结果
        annotated_frame = results[0].plot()
        # 显示图像
        cv2.imshow(winname="YOLO11", mat=annotated_frame)
        # 按ESC退出
        if cv2.waitKey(1) == 27:
            break
    else:
        break
# 释放链接
cap.release()
# 销毁所有窗口
cv2.destroyAllWindows()
```
